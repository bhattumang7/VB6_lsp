; ------------------------------------------------------------
; A Select Case whose every Case has been commented out is still
; a legal, compilable shell. Zero case clauses must parse, with
; the comment/blank lines kept inside the body.
; ------------------------------------------------------------

==================
Round7-A1 Select Case with no Case clause (only comments)
==================

    Select Case x
        'Case "a"
        'commentary
    End Select
End Sub

; ------------------------------------------------------------
; VB6 allows declarations (and other statements) between the
; Select Case selector and the first Case label; they belong to
; the enclosing procedure scope. A leading Dim must not break
; the Select.
; ------------------------------------------------------------

==================
Round7-B1 Dim before the first Case
==================

Sub S()
    Select Case lDay
        Dim iIndex As Integer
        Case 1
            iIndex = 1
    End Select
End Sub
